set(srcs "spi_master_example_main.c"
    "dat_file.c"
    "dat_image.c"
    "prince_dat/src/dat.c"
    "prince_dat/src/autodetect.c"
    "prince_dat/src/auxiliary.c"
    "prince_dat/src/pr_disk_shim.c"
    "prince_dat/src/rle_uncompress.c"
    "prince_dat/src/lzg_uncompress.c"
    )

# Collect every Prince of Persia DAT file to embed into the firmware.
file(GLOB dat_files RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "data/*.DAT")

# Auto-generate a registry mapping each embedded DAT name to its linker
# symbols, from the same list used for EMBED_FILES (single source of truth).
set(dat_registry_src "${CMAKE_CURRENT_BINARY_DIR}/dat_registry.c")
set(dat_registry_body "/* Generated by CMake - do not edit. */\n#include \"dat_registry.h\"\n\n")
set(dat_registry_table "const embedded_dat_t g_embedded_dats[] = {\n")
foreach(dat_file ${dat_files})
    get_filename_component(dat_name "${dat_file}" NAME)     # e.g. TITLE.DAT
    string(MAKE_C_IDENTIFIER "${dat_name}" dat_sym)         # e.g. TITLE_DAT
    string(APPEND dat_registry_body
        "extern const uint8_t _binary_${dat_sym}_start[] asm(\"_binary_${dat_sym}_start\");\n"
        "extern const uint8_t _binary_${dat_sym}_end[]   asm(\"_binary_${dat_sym}_end\");\n")
    string(APPEND dat_registry_table
        "    { \"${dat_name}\", _binary_${dat_sym}_start, _binary_${dat_sym}_end },\n")
endforeach()
string(APPEND dat_registry_table "};\n"
    "const size_t g_embedded_dats_count = sizeof(g_embedded_dats) / sizeof(g_embedded_dats[0]);\n")
file(WRITE "${dat_registry_src}" "${dat_registry_body}\n${dat_registry_table}")

idf_component_register(SRCS ${srcs} "${dat_registry_src}"
                    INCLUDE_DIRS "." "prince_dat/include"
                    PRIV_REQUIRES esp_driver_spi esp_driver_gpio
                    EMBED_FILES ${dat_files})
